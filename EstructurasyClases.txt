// Clases /////////////////////////////////////////////////////////

class Artista {
    var nombre: String
    var apellido: String
    var ultimaObra: String
    var añoMuerte: Int
    var añoNacimiento: Int

    init(_ nombre: String, _ apellido: String, _ ultimaObra: String, _ añoMuerte: Int, _ añoNacimiento: Int) {
        self.nombre = nombre
        self.apellido = apellido
        self.ultimaObra = ultimaObra
        self.añoMuerte = añoMuerte
        self.añoNacimiento = añoNacimiento
    }
}

let pintorPrincipal = Artista("Vincent", "Van Gogh", "Raíces de árbol", 1890, 1853)

// ////////////////////////////////////////////////////////////////

class Museo {
    var lugar: String
    var numeroObras: Double
    var nombreMuseo: String
    var identificador: Double

    init(_ lugar: String, _ numeroObras: Double, _ nombreMuseo: String, _ identificador: Double) {
        self.lugar = lugar
        self.numeroObras = numeroObras
        self.nombreMuseo = nombreMuseo
        self.identificador = identificador
    }
}

let museoPrincipal = Museo("París, Francia", 300000, "Louvre", 88)

// ////////////////////////////////////////////////////////////////

class ObraArte {
    var nombreObra: String
    var añoCreacion: Int
    var nombreArtista: String
    var frase: String  

    init(_ nombreObra: String, _ añoCreacion: Int, _ nombreArtista: String, _ frase: String) {
        self.nombreObra = nombreObra
        self.añoCreacion = añoCreacion
        self.nombreArtista = nombreArtista
        self.frase = frase
    }
}

let obraPrincipal = ObraArte("La noche estrellada", 1889, "Vincent Van Gogh", "Confieso que no sé por qué, pero mirar las estrellas siempre me hace soñar")
